cmake_minimum_required(VERSION 3.21)
project(RaceMate VERSION 0.3.0 LANGUAGES CXX RC)
# Releases are looked up at github.com/<owner>/<repository>/releases/latest.
set(RACEMATE_GITHUB_OWNER "gaggi" CACHE STRING "RaceMate GitHub owner")
set(RACEMATE_GITHUB_REPOSITORY "RaceMate" CACHE STRING "RaceMate GitHub repository")
set(RACEMATE_DEFINITIONS UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN RACEMATE_VERSION="${PROJECT_VERSION}"
  RACEMATE_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} RACEMATE_VERSION_MINOR=${PROJECT_VERSION_MINOR} RACEMATE_VERSION_PATCH=${PROJECT_VERSION_PATCH}
  RACEMATE_GITHUB_OWNER="${RACEMATE_GITHUB_OWNER}" RACEMATE_GITHUB_REPOSITORY="${RACEMATE_GITHUB_REPOSITORY}")
set(RACEMATE_UPDATE_SOURCES src/update/UpdateChecker.cpp src/update/JsonLite.cpp)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(racemate_core src/Core.cpp src/Prediction.cpp src/RaceControl.cpp src/TrackMap.cpp src/PitStop.cpp src/SectorTiming.cpp src/PassingTimes.cpp)
target_include_directories(racemate_core PUBLIC src)
add_executable(RaceMate WIN32 src/main.cpp src/Telemetry.cpp src/TrackMapLoader.cpp ${RACEMATE_UPDATE_SOURCES} resources/app.rc)
target_link_libraries(RaceMate PRIVATE racemate_core comctl32 uxtheme shell32 shlwapi gdi32 gdiplus msimg32 user32 advapi32 winhttp bcrypt)
target_compile_definitions(RaceMate PRIVATE ${RACEMATE_DEFINITIONS})
add_executable(RaceMateDiagnostics WIN32 EXCLUDE_FROM_ALL src/main.cpp src/Telemetry.cpp src/TrackMapLoader.cpp ${RACEMATE_UPDATE_SOURCES} resources/app.rc)
target_link_libraries(RaceMateDiagnostics PRIVATE racemate_core comctl32 uxtheme shell32 shlwapi gdi32 gdiplus msimg32 user32 advapi32 winhttp bcrypt)
target_compile_definitions(RaceMateDiagnostics PRIVATE ${RACEMATE_DEFINITIONS} RACEMATE_DIAGNOSTICS)
include(CTest)
if(BUILD_TESTING)
  add_executable(RaceMateTests tests/CoreTests.cpp tests/PredictionTests.cpp tests/RaceControlTests.cpp tests/ReviewRegressionTests.cpp tests/PitStopTests.cpp tests/SectorTimingTests.cpp tests/PassingTimesTests.cpp src/Telemetry.cpp)
  target_compile_definitions(RaceMateTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(RaceMateTests PRIVATE racemate_core user32)
    if(MSVC)
      # Debug retains the large telemetry/forecast fixtures in main's stack frame.
      target_link_options(RaceMateTests PRIVATE $<$<CONFIG:Debug>:/STACK:8388608>)
    endif()
  add_test(NAME CoreAndTelemetry COMMAND RaceMateTests)
  # A stand-in for iRacing for the live benchmark (RaceMateDiagnostics --benchmark-live).
  add_executable(RaceMateFakeIRacing tools/FakeIRacing.cpp)
  target_compile_definitions(RaceMateFakeIRacing PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(RaceMateUpdateTests tests/UpdateCheckerTests.cpp src/update/JsonLite.cpp)
  target_include_directories(RaceMateUpdateTests PRIVATE src/update)
  target_compile_definitions(RaceMateUpdateTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(RaceMateUpdateTests PRIVATE shell32 winhttp bcrypt)
  add_test(NAME Updater COMMAND RaceMateUpdateTests)
  add_executable(RaceMateTrackMapTests tests/TrackMapTests.cpp src/TrackMapLoader.cpp)
  target_compile_definitions(RaceMateTrackMapTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(RaceMateTrackMapTests PRIVATE racemate_core winhttp)
  add_test(NAME TrackMap COMMAND RaceMateTrackMapTests)
endif()
if(MSVC)
  foreach(target racemate_core RaceMate RaceMateDiagnostics RaceMateTests RaceMateUpdateTests RaceMateTrackMapTests )
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc $<$<CONFIG:Release>:/O1 /Gy /Gw /GF>)
    endif()
  endforeach()
  target_link_options(RaceMate PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /MANIFEST:NO)
  target_link_options(RaceMateDiagnostics PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /MANIFEST:NO)
endif()
# DirectX-only Spout2 subset, pinned to 6b68a5b4fdeca6130b851e5f9366467a844860e5.
set(SPOUT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/third_party/spout)
add_library(racemate_spout STATIC
  ${SPOUT_DIR}/SpoutDX.cpp ${SPOUT_DIR}/SpoutCopy.cpp ${SPOUT_DIR}/SpoutDirectX.cpp
  ${SPOUT_DIR}/SpoutFrameCount.cpp ${SPOUT_DIR}/SpoutSenderNames.cpp
  ${SPOUT_DIR}/SpoutSharedMemory.cpp ${SPOUT_DIR}/SpoutUtils.cpp)
target_compile_definitions(racemate_spout PUBLIC SPOUT_BUILD_STATIC PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(racemate_spout PUBLIC d3d11 dxgi d3d9 version winmm shlwapi)
if(MSVC)
  set_property(TARGET racemate_spout PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
foreach(target RaceMate RaceMateDiagnostics)
  target_sources(${target} PRIVATE src/SpoutOutput.cpp src/SpoutPixels.cpp)
  target_link_libraries(${target} PRIVATE racemate_spout)
endforeach()
# Explicit opt-in hardware smoke test; not part of headless CTest runs.
if(BUILD_TESTING)
  add_executable(RaceMateSpoutTests EXCLUDE_FROM_ALL tests/SpoutOutputTests.cpp src/SpoutOutput.cpp src/SpoutPixels.cpp)
  target_include_directories(RaceMateSpoutTests PRIVATE src)
  target_compile_definitions(RaceMateSpoutTests PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(RaceMateSpoutTests PRIVATE racemate_spout)
  if(MSVC)
    set_property(TARGET RaceMateSpoutTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
if(BUILD_TESTING)
  add_executable(RaceMateSpoutPixelTests tests/SpoutPixelTests.cpp src/SpoutPixels.cpp)
  target_include_directories(RaceMateSpoutPixelTests PRIVATE src)
  target_compile_definitions(RaceMateSpoutPixelTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(RaceMateSpoutPixelTests PRIVATE gdi32 user32)
  if(MSVC)
    set_property(TARGET RaceMateSpoutPixelTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(RaceMateSpoutPixelTests PRIVATE /W4 /utf-8 /EHsc)
  endif()
  add_test(NAME SpoutPixels COMMAND RaceMateSpoutPixelTests)
endif()
