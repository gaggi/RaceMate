cmake_minimum_required(VERSION 3.21)
project(RaceMate VERSION 0.4.36 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(racemate_core src/Core.cpp src/Prediction.cpp src/RaceControl.cpp)
target_include_directories(racemate_core PUBLIC src)
add_executable(RaceMate WIN32 src/main.cpp src/Telemetry.cpp resources/app.rc)
target_link_libraries(RaceMate PRIVATE racemate_core comctl32 shell32 gdi32 gdiplus user32 advapi32)
target_compile_definitions(RaceMate PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(RaceMateDiagnostics WIN32 EXCLUDE_FROM_ALL src/main.cpp src/Telemetry.cpp resources/app.rc)
target_link_libraries(RaceMateDiagnostics PRIVATE racemate_core comctl32 shell32 gdi32 gdiplus user32 advapi32)
target_compile_definitions(RaceMateDiagnostics PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN RACEMATE_DIAGNOSTICS)
include(CTest)
if(BUILD_TESTING)
  add_executable(RaceMateTests tests/CoreTests.cpp tests/PredictionTests.cpp tests/RaceControlTests.cpp tests/ReviewRegressionTests.cpp src/Telemetry.cpp)
  target_compile_definitions(RaceMateTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(RaceMateTests PRIVATE racemate_core user32)
  add_test(NAME CoreAndTelemetry COMMAND RaceMateTests)
endif()
if(MSVC)
  foreach(target racemate_core RaceMate RaceMateDiagnostics RaceMateTests)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc $<$<CONFIG:Release>:/O1 /Gy /Gw /GF>)
    endif()
  endforeach()
  target_link_options(RaceMate PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /MANIFEST:NO)
  target_link_options(RaceMateDiagnostics PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /MANIFEST:NO)
endif()
